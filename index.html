CREATE TABLE trades (
  id            uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  symbol        text,
  direction     text,
  status        text,
  lots          numeric,
  entry         numeric,
  exit          numeric,
  sl            numeric,
  tp            numeric,
  gross_pnl     numeric,
  commission    numeric,
  risk          numeric,
  pnl           numeric,
  rr            numeric,
  session       text,
  setup         text,
  typ           text DEFAULT 'Live',
  stadium       text DEFAULT 'Challenge',
  notes         text,
  screenshot_entry text,
  screenshot_exit  text,
  date_open     timestamptz,
  date_close    timestamptz,
  created_at    timestamptz DEFAULT now()
);

-- Row Level Security
ALTER TABLE trades ENABLE ROW LEVEL SECURITY;

-- Policy: voller Zugriff über anon key (Single-User)
CREATE POLICY "allow_all" ON trades
  FOR ALL TO anon USING (true) WITH CHECK (true);
